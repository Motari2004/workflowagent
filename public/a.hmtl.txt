function fitCanvas() {
    const cvs = canvas.canvas, rect = cvs.getBoundingClientRect();
    const w = Math.max(1, Math.round(rect.width)), h = Math.max(1, Math.round(rect.height));
    if (cvs.width !== w || cvs.height !== h) {
        cvs.width = w; cvs.height = h;
        canvas.setDirty(true, true); canvas.draw(true);
    }

canvas.getNodeOnPos = function(x,y,list,margin) {
    const nodes = list || this.graph._nodes; margin = margin || 0;
    for (let i = nodes.length-1; i >= 0; i--) {
        const n = nodes[i]; if (!n) continue;
        const s = n.size;
        if (x>=n.pos[0]-margin && y>=n.pos[1]-margin && x<=n.pos[0]+s[0]+margin && y<=n.pos[1]+s[1]+margin) return n;
    }
    return null;
};

// Add node
let spawnOffset = 0;
function addNode(type) {
    try {
        fitCanvas();
        const node = LiteGraph.createNode(type);
        if (!node) throw new Error(`Unknown: ${type}`);
        const rect = canvas.canvas.getBoundingClientRect(), ds = canvas.ds;
        const cx = (rect.width/2 - ds.offset[0]) / ds.scale;
        const cy = (rect.height/2 - ds.offset[1]) / ds.scale;
        const step = spawnOffset % 10;
        node.pos = [cx - 100 + step*30, cy - 50 + step*25];
        spawnOffset++;
        graph.add(node);
        canvas.selected_nodes = {};
        canvas.setDirty(true, true);
        toast(`Added: ${node.title}`, 'info', 1200);
    } catch (e) { toast(`Error: ${e.message}`, 'error'); console.error(e); }
}

// Build payload

// ============================================================
// Safe JSON helpers — must exist before buildPayload() runs
// ============================================================
function safeJsonParse(str) {
    if (str === null || str === undefined) return null;
    if (typeof str === 'object') return str;  // already parsed
    if (typeof str !== 'string') return str;
    const trimmed = str.trim();
    if (trimmed === '') return null;
    try {
        return JSON.parse(trimmed);
    } catch (e) {
        console.warn('safeJsonParse: invalid JSON, returning raw string:', trimmed.slice(0, 100));
        return trimmed;  // fallback — pass raw string through
    }
}

function safeJsonStringify(val) {
    try {
        return JSON.stringify(val);
    } catch (e) {
        console.warn('safeJsonStringify failed:', e);
        return String(val);
    }
}

function buildPayload() {
    const visited = new Set(), ordered = [];
    function visit(n) {
        if (!n || visited.has(n)) return;
        visited.add(n);
        (n.inputs||[]).forEach(input => {
            if (input.link != null) {
                const link = graph.links[input.link];
                if (link) visit(graph.getNodeById(link.origin_id));
            }
        });
        ordered.push(n);
    }
    const hasWires = Object.keys(graph.links || {}).length > 0;
    if (hasWires) {
        const sinks = graph._nodes.filter(n => !n.outputs || n.outputs.every(o => o.link == null));
        (sinks.length ? sinks : graph._nodes).forEach(visit);
    } else graph._nodes.forEach(visit);

    // Only sort by stage if there are NO wires.
    // When wires exist, trust the topological order — it already follows edges.
    if (!hasWires) {
        const stageOrder = {
            'trigger/webhook':0, 'market/data':0,
            'logic/filter':1, 'logic/math':1, 'logic/compare':1, 'logic/json-extract':1, 'logic/code':1,
            'action/http':2, 'tool/integration':2, 'output/console':3,
        };
        ordered.sort((a,b) => (stageOrder[a.type]??99) - (stageOrder[b.type]??99));
    }

    const payload = [];
    for (const n of ordered) {
        const d = n.properties || {};
        switch (n.type) {
            case 'trigger/webhook': payload.push({type:'webhook-input', data:{}}); break;
            case 'market/data': payload.push({type:'market-data', data:{coin:d.coin||'bitcoin'}}); break;
            case 'logic/filter': payload.push({type:'filter', data:{keyword:d.keyword||'Price'}}); break;
            case 'logic/math': payload.push({type:'math', data:{op:d.op||'multiply', value:Number(d.value)||1}}); break;
            case 'logic/compare': payload.push({type:'compare', data:{operator:d.operator||'>', threshold:Number(d.threshold)||0}}); break;
            case 'logic/json-extract': payload.push({type:'json-extract', data:{path:d.path||''}}); break;
            case 'tool/integration': payload.push({type:'tool', data:{service:d.service||'http-custom', message:d.message||''}}); break;
            case 'action/http': payload.push({type:'http-request', data:{
                url: d.url || '',
                method: d.method || 'GET',
                body: d.body ? safeJsonParse(d.body) : null,
                headers: d.headers ? safeJsonParse(d.headers) : {}
            }}); break;
            case 'logic/delay': payload.push({type:'delay', data:{seconds: Number(d.seconds) || 5}}); break;
            case 'logic/code': payload.push({type:'code', data:{lang:d.lang||'javascript', code:d.code||''}}); break;
            case 'output/console': payload.push({type:'console', data:{}}); break;
        }
    }
    return payload;
}

// Logs
function classifyLog(m) {
    const s = m.toLowerCase();
    if (s.includes('error') || s.includes('reject') || s.includes('fail')) return 'error';
    if (s.includes('success') || s.includes('match') || s.includes('validat') || s.includes('pass')) return 'success';
    if (s.includes('skip') || s.includes('warn')) return 'warn';
    return 'info';
}
function renderLogs(logs, result) {
    const logDiv = document.getElementById('log');
    logDiv.innerHTML = '';
    (logs||[]).forEach(l => {
        const d = document.createElement('div');
        d.className = `log-entry ${classifyLog(l)}`;
        d.textContent = l; logDiv.appendChild(d);
    });
    if (result) {
        const r = document.createElement('div');
        r.className = 'result-card';
        r.innerHTML = `<i class="fas fa-check-circle mr-2"></i>${result}`;
        logDiv.appendChild(r);
    }
    logDiv.scrollTop = logDiv.scrollHeight;
}

// Run
let isRunning = false;
async function run() {
    if (isRunning) return;
    const btn = document.getElementById('runBtn'), txt = document.getElementById('runBtnText'), log = document.getElementById('log');
    const payload = buildPayload();
    if (payload.length === 0) { toast('No nodes!','error'); return; }
    isRunning = true; btn.disabled = true; txt.textContent = 'EXECUTING...';
    btn.querySelector('i').className = 'fas fa-spinner fa-spin';
    log.innerHTML = '<div class="log-entry info">Sending...</div>';
    try {
        const wfName = document.getElementById('workflowName').value.trim() || 'manual';
        const res = await fetch('http://localhost:5000/run', {
            method:'POST', headers:{'Content-Type':'application/json'},
            body: JSON.stringify({nodes:payload, workflow_name:wfName}),
        });
        if (!res.ok) throw new Error(`Server ${res.status}`);
        const data = await res.json();
        renderLogs(data.logs, data.data);
        if (data.status === 'success' && data.data) toast('Workflow complete!','success');
        else toast('Finished','info');
    } catch (e) {
        log.innerHTML = `<div class="log-entry error">${e.message}</div>`;
        toast(e.message, 'error');
    } finally {
        isRunning = false; btn.disabled = false; txt.textContent = 'EXECUTE WORKFLOW';
        btn.querySelector('i').className = 'fas fa-play';
    }
}

async function openEditCredsModal(service) {
    try {
        const tool = TOOLS[service];
        if (!tool) { toast("Unknown tool: " + service, 'error'); return; }

        // Fetch existing credentials (safe — 404 returns empty object)
        let existing = {};
        try {
            const r = await fetch(`http://localhost:5000/credentials/${service}`);
            if (r.ok) existing = await r.json();
        } catch (_) { /* not configured yet — that's fine */ }

        // Remove any prior modal
        document.querySelectorAll('.modal-overlay').forEach(m => m.remove());

        const fieldsHtml = tool.creds.length > 0
            ? tool.creds.map(k => {
                const val = (existing[k] || '').toString().replace(/"/g, '&quot;');
                return `<label>${k.replace(/_/g, ' ')}</label><input name="${k}" value="${val}" autocomplete="off" />`;
            }).join('')
            : '<p class="text-xs text-slate-500">This tool requires no credentials.</p>';

        const overlay = document.createElement('div');
        overlay.className = 'modal-overlay';
        overlay.addEventListener('click', (e) => { if (e.target === overlay) overlay.remove(); });

        overlay.innerHTML = `
            <div class="modal">
                <h2><i class="${tool.brand ? 'fab' : 'fas'} ${tool.icon} mr-2" style="color:${tool.color}"></i>${tool.label} Credentials</h2>
                <form id="credsForm">
                    ${fieldsHtml}
                    <div class="modal-buttons">
                        <button type="button" class="modal-btn secondary" id="cancelCreds">Cancel</button>
                        <button type="submit" class="modal-btn primary">Save</button>
                    </div>
                </form>
            </div>`;
        document.body.appendChild(overlay);

        document.getElementById('cancelCreds').addEventListener('click', () => overlay.remove());

        document.getElementById('credsForm').addEventListener('submit', async (e) => {
            e.preventDefault();
            const form = e.target;
            const creds = {};
            tool.creds.forEach(k => { creds[k] = form.elements[k].value; });
            try {
                const res = await fetch(`http://localhost:5000/credentials/${service}`, {
                    method: 'PUT',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify(creds),
                });
                if (res.ok) {
                    toast(`${tool.label} saved`, 'success');
                    overlay.remove();
                    // Re-open main credentials panel with fresh badges
                    setTimeout(openCredentialsPanel, 100);
                } else {
                    toast('Save failed: ' + res.status, 'error');
                }
            } catch (err) {
                toast('Save error: ' + err.message, 'error');
            }
        });
    } catch (e) {
        console.error("openEditCredsModal failed:", e);
        toast("Failed to open editor: " + e.message, 'error', 4000);
    }
}

                   '<div class="ai-code-header">' +
                     '<span class="ai-code-lang">' + langLabel + '</span>' +
                     '<button class="ai-copy-btn" data-code="' + encoded + '" title="Copy to clipboard">' +
                       '<span>Copy</span>' +
                     '</button>' +
                   '</div>' +
                   '<pre class="ai-code-content">' + cleanCode + '</pre>' +
                 '</div>';
        });

        out = out.replace(/`([^`\n]+)`/g, '<code>$1</code>');

        const parts = out.split(/(<pre[\s\S]*?<\/pre>|<div class="ai-code-block"[\s\S]*?<\/div>)/);
        for (let i = 0; i < parts.length; i++) {
            if (!parts[i].startsWith('<pre') && !parts[i].startsWith('<div class="ai-code-block"')) {
                parts[i] = parts[i].replace(/\n/g, '<br>');
            }
        }
        out = parts.join('');

        return out;
    }
    
    function appendMessage(role, text) {
        const el = document.createElement('div');
        el.className = 'ai-msg ' + role;
        el.innerHTML = role === 'bot' ? renderMarkdown(text) : String(text).replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;');
        body.appendChild(el);
        body.scrollTop = body.scrollHeight;
    }
    
    function appendThinking() {
        const el = document.createElement('div');
        el.className = 'ai-thinking';
        el.textContent = 'Thinking';
        body.appendChild(el);
        body.scrollTop = body.scrollHeight;
        return el;
    }
    
    function renderHistory() {
        body.innerHTML = '';
        if (history.length === 0) {
            appendMessage('bot', "Hi! I'm your workflow assistant. Ask me anything.");
        } else {
            history.forEach(h => appendMessage(h.role, h.text));
        }
    }
    
    // ---- Send ----
    async function sendMessage(text) {
        text = String(text || '').trim();
        if (!text || busy) return;
        busy = true;
        if (sendBtn) sendBtn.disabled = true;
        
        appendMessage('user', text);
        history.push({ role: 'user', text });
        const thinking = appendThinking();
        
        try {
            const res = await fetch('http://localhost:5000/ai-chat', {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({
                    message: text,
                    history: history.slice(-10),
                    model: (modelSelector && modelSelector.value) || 'gemini-flash'
                })
            });
            const data = await res.json();
            thinking.remove();
            
            if (!res.ok || data.error) {
                const errText = data.error || ('Server error ' + res.status);
                appendMessage('bot', 'Warning: ' + errText);
                history.push({ role: 'bot', text: 'Warning: ' + errText });
            } else {
                appendMessage('bot', data.reply || '(empty)');
                history.push({ role: 'bot', text: data.reply || '(empty)' });
            }
            saveHistory();
        } catch (err) {
            thinking.remove();
            appendMessage('bot', 'Network error: ' + err.message);
        } finally {
            busy = false;
            if (sendBtn) sendBtn.disabled = false;
            input.focus();
        }
    }
    
    // ---- Input handlers ----
    function handleSendFromInput() {
        const v = input.value.trim();
        if (!v) return;
        input.value = '';
        input.style.height = 'auto';
        sendMessage(v);
    }
    
    input.addEventListener('keydown', (e) => {
        if (e.key === 'Enter' && !e.shiftKey) {
            e.preventDefault();
            handleSendFromInput();
        }
    });
    
    input.addEventListener('input', () => {
        input.style.height = 'auto';
        input.style.height = Math.min(100, input.scrollHeight) + 'px';
    });
    
    if (sendBtn) {
        sendBtn.addEventListener('click', (e) => {
            e.stopPropagation();
            handleSendFromInput();
        });
    }
    
    // ---- Toggle ----
    function togglePanel(e) {
        if (e) {
            if (e.target && e.target.closest) {
                if (e.target.closest('#aiClearBtn')) return;
                if (e.target.closest('#aiModelSelector')) return;
            }
        }
        if (modelSelector && document.activeElement === modelSelector) return;
        panel.classList.toggle('collapsed');
        const isCollapsed = panel.classList.contains('collapsed');
        if (!isCollapsed) setTimeout(() => input.focus(), 100);
    }
    header.addEventListener('click', togglePanel);
    
    // ---- Clear ----
    if (clearBtn) {
        clearBtn.addEventListener('click', (e) => {
            e.stopPropagation();
            if (!confirm('Clear the entire conversation?')) return;
            history = [];
            saveHistory();
            renderHistory();
        });
    }
    
    // ---- Quick actions ----
    quickBtns.forEach(btn => {
        btn.addEventListener('click', () => {
            if (panel.classList.contains('collapsed')) {
                panel.classList.remove('collapsed');
            }
            sendMessage(btn.dataset.prompt);
        });
    });
    
    // ---- Model selector ----
    async function loadModels() {
        if (!modelSelector) return;
        try {
            const res = await fetch('http://localhost:5000/ai-models');
            const data = await res.json();
            modelSelector.innerHTML = '';
            data.models.forEach(m => {
                const opt = document.createElement('option');
                opt.value = m.key;
                opt.textContent = m.label + (m.available ? '' : ' (not configured)');
                opt.disabled = !m.available;
                modelSelector.appendChild(opt);
            });
            const saved = localStorage.getItem(MODEL_STORAGE_KEY);
            if (saved && Array.from(modelSelector.options).some(o => o.value === saved && !o.disabled)) {
                modelSelector.value = saved;
            } else {
                const firstAvail = Array.from(modelSelector.options).find(o => !o.disabled);
                if (firstAvail) modelSelector.value = firstAvail.value;
            }
            console.log('[models] Loaded', data.models.length, 'models');
        } catch (e) {
            console.warn('[models] Could not load:', e);
        }
    }
    
    if (modelSelector) {
        modelSelector.addEventListener('change', () => {
            localStorage.setItem(MODEL_STORAGE_KEY, modelSelector.value);
        });
        ['click', 'mousedown', 'mouseup'].forEach(evt => {
            modelSelector.addEventListener(evt, (e) => e.stopPropagation());
        });
        loadModels();
    }
    

    // ============ COPY BUTTON HANDLER ============
    document.addEventListener('click', async (e) => {
        const btn = e.target.closest('.ai-copy-btn');
        if (!btn) return;

        const encoded = btn.dataset.code;
        if (!encoded) return;

        const code = decodeURIComponent(encoded);

        try {
            await navigator.clipboard.writeText(code);
            const originalHTML = btn.innerHTML;
            btn.innerHTML = '<span>Copied</span>';
            btn.style.color = '#22c55e';
            setTimeout(() => {
                btn.innerHTML = originalHTML;
                btn.style.color = '';
            }, 1500);
        } catch (err) {
            const textarea = document.createElement('textarea');
            textarea.value = code;
            textarea.style.position = 'fixed';
            textarea.style.opacity = '0';
            document.body.appendChild(textarea);
            textarea.select();
            try {
                document.execCommand('copy');
                btn.innerHTML = '<span>Copied</span>';
                btn.style.color = '#22c55e';
                setTimeout(() => {
                    btn.innerHTML = '<span>Copy</span>';
                    btn.style.color = '';
                }, 1500);
            } catch (e2) {
                alert('Copy failed. Select and copy manually.');
            }
            document.body.removeChild(textarea);
        }
    });
    // ============ END COPY BUTTON HANDLER ============

        // ---- Initial render ----
    renderHistory();
    console.log('[ai] AI panel ready');
})();
